	RTLSDR=yes \

#!/bin/sh
#
# Start mlat-client for ADSBexchange
#

NAME="mlat-client-adsbx"
DAEMON="/usr/bin/mlat-client"
PIDFILE="/var/run/$NAME.pid"

ARGS="--input-type dump1090 \
      --input-connect 127.0.0.1:30005 \
      --lat 47.249544584520905 \
      --lon -2.262069274883702 \
      --alt 45m \
      --user Rfontenay_ADSBX \
      --server feed.adsbexchange.com:31090 \
      --no-udp \
      --results beast,connect,127.0.0.1:30004"

case "$1" in
  start)
        echo -n "Starting $NAME: "
        start-stop-daemon -S -q -b -m -p $PIDFILE -x $DAEMON -- $ARGS
        echo "OK"
        ;;
  stop)
        echo -n "Stopping $NAME: "
        start-stop-daemon -K -q -p $PIDFILE
        echo "OK"
        rm -f $PIDFILE
        ;;
  restart|reload)
        $0 stop
        sleep 1
        $0 start
        ;;
  *)
        echo "Usage: $0 {start|stop|restart|reload}"
        exit 1
        ;;
esac

exit 0

#!/bin/sh

DAEMON="readsb"
BINARY="/usr/bin/readsb"

# Identifiant unique pour éviter les doublons sur les cartes communautaires
SITE_UUID="b8c4d720-a6f3-4e89-91b5-3c82d41ef9a1"

# Arguments finaux sans l'option biastee
ARGS="--device-type rtlsdr \
      --net \
      --fix \
      --dcfilter \
      --lat 47.249544584520905 \
      --lon -2.262069274883702 \
      --net-ri-port 0 \
      --net-ro-port 30002 \
      --net-bi-port 30004 \
      --net-bo-port 30005 \
      --write-json /tmp/readsb \
      --net-connector feed.adsb.fi,30004,beast_reduce_out,uuid=$SITE_UUID \
      --net-connector feed.adsbexchange.com,30004,beast_reduce_out"

case "$1" in
    start)
        echo -n "Starting $DAEMON: "

        # PLAN B : On active l'alim antenne via l'outil externe juste avant readsb
        /usr/bin/rtl_biast -b 1 > /dev/null 2>&1
        sleep 1

        start-stop-daemon -S -b -x $BINARY -- $ARGS
        [ $? -eq 0 ] && echo "OK" || echo "FAIL"
        ;;
    stop)
        echo -n "Stopping $DAEMON: "
        start-stop-daemon -K -x $BINARY
        [ $? -eq 0 ] && echo "OK" || echo "FAIL"
        ;;
    restart)
        $0 stop
        sleep 1
        $0 start
        ;;
    *)
        echo "Usage: $0 {start|stop|restart}"
        exit 1
        ;;
esac

exit 0

# Start mlat-client for adsb.fi
#

DAEMON="/usr/bin/mlat-client"
PIDFILE="/var/run/$NAME.pid"

# Tes paramètres exacts validés
ARGS="--input-type dump1090 \
      --input-connect 127.0.0.1:30005 \
      --lat 47.249544584520905 \
      --lon -2.262069274883702 \
      --alt 45m \
      --user Rfontenay \

      --no-udp \
      --results beast,connect,127.0.0.1:30004"

case "$1" in
  start)
        echo -n "Starting $NAME: "
        # -b lance en tâche de fond (background)

        echo -n "Stopping $NAME: "

        echo "OK"
        # On supprime le PID file proprement
        rm -f $PIDFILE
        ;;
  restart|reload)
        $0 stop
        sleep 1
        $0 start
        ;;
  *)
        echo "Usage: $0 {start|stop|restart|reload}"
        exit 1
        ;;
esac

exit 0

# Modules nécessaires
server.modules += ( "mod_alias", "mod_setenv" )

# --- CONFIGURATION DU ROOT ---
# On définit le dossier de tar1090 comme racine du serveur
server.document-root        = "/var/www/tar1090"
server.port                 = 80
index-file.names            = ( "index.html" )

# Types MIME essentiels
mimetype.assign = (
  ".html" => "text/html",
  ".css"  => "text/css",
  ".js"   => "application/javascript",
  ".json" => "application/json",
  ".png"  => "image/png",
  ".svg"  => "image/svg+xml"
)

# --- LIAISON DES DONNÉES TEMPS RÉEL ---
# Comme on est à la racine, tar1090 va chercher ses fichiers dans /data/
# On pointe donc /data/ vers ton dossier en RAM (/tmp/readsb)
alias.url += ( "/data/" => "/tmp/readsb/" )

# --- OPTIMISATION FLUIDITÉ ---
# On interdit le cache sur les JSON pour éviter que les avions ne "sautent"
$HTTP["url"] =~ "^/data/" {
    setenv.add-response-header = ( "Cache-Control" => "no-cache, no-store, must-revalidate" )
}
